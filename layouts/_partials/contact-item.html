{{- /* One contact or link from data/profile.yaml. Both use the same fields: name optional label, shown as "name: text" (or alone when there is no text) icon optional icon name (see assets/icons) text the value, e.g. an email address or handle ("info" is accepted too) url optional link; the text (or the name, when there is no text) becomes the link A plain string is shown as Markdown. */ -}} {{- $c := . }} {{- if reflect.IsMap $c }} {{- $text := $c.text | default $c.info }} {{- $link := $text | default $c.name }} {{ partial "icon.html" $c.icon }} {{- if and $text $c.name }}{{ $c.name }}: {{ end }} {{- /* Plain text inside links: Markdown would turn an email address into a second, nested link. */ -}} {{- with $c.url }}{{ $link }}{{ else }}{{ $link | markdownify }}{{ end -}} {{- else }} {{ $c | markdownify }} {{- end -}}
