{{- /* One contact or link from data/profile.yaml. Both use the same fields: name optional label, shown as "name: text" (or alone when there is no text) icon optional icon name (see assets/icons) text the value, e.g. an email address or handle ("info" is accepted too) url optional link; the text (or the name, when there is no text) becomes the link A plain string is shown as Markdown. The name is wrapped in .contact-name so that, with params.hideContactNames, the web page can show the icon in its place (see main.css); the printable CV always shows it. */ -}} {{- $c := . }} {{- if reflect.IsMap $c }} {{- $text := $c.text | default $c.info }} {{- $icon := partial "icon.html" $c.icon }} {{- if $text }} {{- $icon }} {{- with $c.name }}{{ . }}: {{ end }} {{- /* Plain text inside links: Markdown would turn an email address into a second, nested link. */ -}} {{- with $c.url }}{{ $text }}{{ else }}{{ $text | markdownify }}{{ end }} {{- else }} {{- /* Only a name: it is the link, so the icon goes inside the link too. */ -}} {{- with $c.url }}{{ $icon }}{{ $c.name }} {{- else }}{{ $icon }}{{ $c.name | markdownify }}{{ end }} {{- end -}} {{- else }} {{ $c | markdownify }} {{- end -}}
