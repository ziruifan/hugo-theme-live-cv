{{- /* Collaborators of a page as HTML. Names found in data/people.* are linked to their url. */ -}} {{- $people := (partial "func/items.html" site.Data.people).items }} {{- $out := slice }} {{- range .Params.collaborators }} {{- $name := . }} {{- $url := "" }} {{- if reflect.IsMap . }}{{ $name = .name }}{{ $url = .url }}{{ end }} {{- if not $url }} {{- range $people }}{{ if eq .name $name }}{{ $url = .url }}{{ end }}{{ end }} {{- end }} {{- if $url }} {{- $out = $out | append (printf "%s" $url (htmlEscape $name)) }} {{- else }} {{- $out = $out | append (htmlEscape $name) }} {{- end }} {{- end }} {{- return (delimit $out ", " | safeHTML) -}}
