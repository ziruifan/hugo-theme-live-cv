{{- with $prefix }}{{ . }}{{ if $parts }} | {{ end }}{{ end }} {{- range $i, $k := $parts }}{{ if $i }} · {{ end }}{{ if and $link $k.url }}{{ $k.text }}{{ else }}{{ $k.text }}{{ end }}{{ end -}}
